   160→   160→function useMediaQuery(query) {
   161→   161→  const [matches, setMatches] = useState(false);
   162→   162→  useEffect(() => {
   163→   163→    if (typeof window === "undefined" || !window.matchMedia) return;
   164→   164→    const mql = window.matchMedia(query);
   165→   165→    const update = () => setMatches(mql.matches);
   166→   166→    update();
   167→   167→    if (mql.addEventListener) mql.addEventListener("change", update);
   168→   168→    else mql.addListener(update);
   169→   169→    return () => {
   170→   170→      if (mql.removeEventListener) mql.removeEventListener("change", update);
   171→   171→      else mql.removeListener(update);
   172→   172→    };
   173→   173→  }, [query]);
   174→   174→  return matches;
   175→   175→}

   222→   222→function MagneticButton({ children, disabled, reduceMotion, ...rest }) {
   223→   223→  const ref = useRef(null);
   224→   224→  const [offset, setOffset] = useState({ x: 0, y: 0 });
   225→   225→
   226→   226→  const onMove = useCallback(
   227→   227→    (e) => {
   228→   228→      if (reduceMotion || !ref.current) return;
   229→   229→      const r = ref.current.getBoundingClientRect();
   230→   230→      const cx = r.left + r.width / 2;
   231→   231→      const cy = r.top + r.height / 2;
   232→   232→      const dx = e.clientX - cx;
   233→   233→      const dy = e.clientY - cy;
   234→   234→      const max = 4;
   235→   235→      setOffset({
   236→   236→        x: Math.max(-max, Math.min(max, dx * 0.12)),
   237→   237→        y: Math.max(-max, Math.min(max, dy * 0.3)),
   238→   238→      });
   239→   239→    },
   240→   240→    [reduceMotion]
   241→   241→  );
   242→   242→
   243→   243→  const reset = useCallback(() => setOffset({ x: 0, y: 0 }), []);
   244→   244→
   245→   245→  return (
   246→   246→    <button
   247→   247→      ref={ref}
   248→   248→      type="submit"
   249→   249→      disabled={disabled}
   250→   250→      onMouseMove={onMove}
   251→   251→      onMouseLeave={reset}
   252→   252→      className="b13-cta"
   253→   253→      style={{ transform: `translate(${offset.x}px, ${offset.y}px)` }}
   254→   254→      {...rest}
   255→   255→    >
   256→   256→      <span>{children}</span>
   257→   257→      <ArrowLeft className="b13-cta-icon" size={16} strokeWidth={1.75} />
   258→   258→    </button>
   259→   259→  );
   260→   260→}

   559→   559→function OtpView({ phone, onBack, onVerified, reduceMotion }) {
   560→   560→  const [digits, setDigits] = useState(["", "", "", "", "", ""]);
   561→   561→  const [status, setStatus] = useState("sent"); // idle | sending | sent | verifying | verified | error | expired
   562→   562→  const [seconds, setSeconds] = useState(119);
   563→   563→  const [errorShake, setErrorShake] = useState(false);
   564→   564→  const [converging, setConverging] = useState(false);
   565→   565→  const inputsRef = useRef([]);
   566→   566→
   567→   567→  const maskedPhone = phone && phone.length >= 2 ? `09••••••${phone.slice(-2)}` : "09•••••••••";
   568→   568→
   569→   569→  useEffect(() => {
   570→   570→    const t = setInterval(() => {
   571→   571→      setSeconds((s) => {
   572→   572→        if (s <= 1) {
   573→   573→          clearInterval(t);
   574→   574→          setStatus((prev) => (prev === "verifying" || prev === "verified" ? prev : "expired"));
   575→   575→          return 0;
   576→   576→        }
   577→   577→        return s - 1;
   578→   578→      });
   579→   579→    }, 1000);
   580→   580→    return () => clearInterval(t);
   581→   581→  }, []);
   582→   582→
   583→   583→  useEffect(() => {
   584→   584→    // Focus the first cell on mount.
   585→   585→    const first = inputsRef.current[0];
   586→   586→    if (first) first.focus();
   587→   587→  }, []);
   588→   588→
   589→   589→  /* -- DEMO ONLY --------------------------------------------------
   590→   590→     In production, replace this with:
   591→   591→       POST /api/auth/verify-otp   { phone, code }
   592→   592→     and read the resulting session/token + error/rate-limit state
   593→   593→     from the response. OTP generation and checking must never
   594→   594→     happen in the browser. */
   595→   595→  const verify = useCallback(
   596→   596→    (code) => {
   597→   597→      setStatus("verifying");
   598→   598→      setTimeout(() => {
   599→   599→        if (code.length === 6) {
   600→   600→          setStatus("verified");
   601→   601→          // Signature moment: the six cells converge toward the centre
   602→   602→          // and hand off to the success ring before the parent swaps views.
   603→   603→          setTimeout(() => {
   604→   604→            setConverging(true);
   605→   605→            setTimeout(onVerified, reduceMotion ? 0 : 560);
   606→   606→          }, 150);
   607→   607→        } else {
   608→   608→          setStatus("error");
   609→   609→          setErrorShake(true);
   610→   610→          setTimeout(() => setErrorShake(false), 420);
   611→   611→        }
   612→   612→      }, 700);
   613→   613→    },
   614→   614→    [onVerified, reduceMotion]
   615→   615→  );
   616→   616→
   617→   617→  const setDigitAt = (i, val) => {
   618→   618→    setDigits((prev) => {
   619→   619→      const next = [...prev];
   620→   620→      next[i] = val;
   621→   621→      return next;
   622→   622→    });
   623→   623→  };
   624→   624→
   625→   625→  const handleChange = (i, e) => {
   626→   626→    const raw = e.target.value.replace(/[^0-9]/g, "");
   627→   627→    if (!raw) {
   628→   628→      setDigitAt(i, "");
   629→   629→      return;
   630→   630→    }
   631→   631→    const char = raw[raw.length - 1];
   632→   632→    setDigitAt(i, char);
   633→   633→    if (i < 5) {
   634→   634→      const nextInput = inputsRef.current[i + 1];
   635→   635→      if (nextInput) nextInput.focus();
   636→   636→    } else {
   637→   637→      const code = digits.map((d, idx) => (idx === i ? char : d)).join("");
   638→   638→      if (code.length === 6) verify(code);
   639→   639→    }
   640→   640→  };
   641→   641→
   642→   642→  const handleKeyDown = (i, e) => {
   643→   643→    if (e.key === "Backspace") {
   644→   644→      if (digits[i]) {
   645→   645→        setDigitAt(i, "");
   646→   646→      } else if (i > 0) {
   647→   647→        const prevInput = inputsRef.current[i - 1];
   648→   648→        if (prevInput) prevInput.focus();
   649→   649→        setDigitAt(i - 1, "");
   650→   650→      }
   651→   651→    } else if (e.key === "ArrowRight") {
   652→   652→      const t = inputsRef.current[i + 1];
   653→   653→      if (t) t.focus();
   654→   654→    } else if (e.key === "ArrowLeft") {
   655→   655→      const t = inputsRef.current[i - 1];
   656→   656→      if (t) t.focus();
   657→   657→    }
   658→   658→  };
   659→   659→
   660→   660→  const handlePaste = (e) => {

   668→   668→    const lastIdx = Math.min(chars.length, 6) - 1;
   669→   669→    const target = inputsRef.current[lastIdx >= 0 ? lastIdx : 0];
   670→   670→    if (target) target.focus();

   679→   679→    if (first) first.focus();
   680→   680→  };

   682→   682→  return (

   687→   687→
   688→   688→      <div className="b13-panel-head">
   689→   689→        <span className="b13-eyebrow">SECURE ACCESS</span>
   690→   690→        <h1 className="b13-display b13-panel-title">Verify.</h1>
   691→   691→        <p className="b13-fa b13-panel-sub">
   692→   692→          کد تأیید ۶ رقمی برای شماره
   693→   693→          <br />
   694→   694→          <span dir="ltr" className="b13-phone-chip">
   695→   695→            {maskedPhone}
   696→   696→          </span>{" "}
   697→   697→          ارسال شد.
   698→   698→        </p>
   699→   699→      </div>
   700→   700→
   701→   701→      <div
   702→   702→        className={"b13-otp-row" + (errorShake ? " b13-otp-row--shake" : "")}
   703→   703→        onPaste={handlePaste}
   704→   704→        role="group"
   705→   705→        aria-label="کد تأیید شش رقمی"
   706→   706→      >
   707→   707→        {digits.map((d, i) => {
   708→   708→          const dist = i - 2.5;
   709→   709→          const convergeStyle = converging
   710→   710→            ? {
   711→   711→                transform: `translateX(${dist * -30}px) scale(0.35)`,
   712→   712→                opacity: 0,
   713→   713→                transitionDelay: `${Math.abs(dist) * 40}ms`,
   714→   714→                transitionDuration: "480ms",
   715→   715→              }
   716→   716→            : undefined;
   717→   717→          return (
   718→   718→            <input
   719→   719→              key={i}
   720→   720→              ref={(el) => (inputsRef.current[i] = el)}
   721→   721→              type="text"
   722→   722→              inputMode="numeric"
   723→   723→              maxLength={1}
   724→   724→              readOnly={converging}
   725→   725→              value={d}
   726→   726→              onChange={(e) => handleChange(i, e)}
   727→   727→              onKeyDown={(e) => handleKeyDown(i, e)}

   730→   730→                "b13-otp-cell" +
   731→   731→                (d ? " b13-otp-cell--filled" : "") +
   732→   732→                (status === "error" ? " b13-otp-cell--error" : "")
   733→   733→              }
   734→   734→              aria-label={`رقم ${i + 1} از ۶`}

   740→   740→      <div className="b13-otp-status" aria-live="polite">
