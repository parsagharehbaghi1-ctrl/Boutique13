   160→function useMediaQuery(query) {
   161→  const [matches, setMatches] = useState(false);
   162→  useEffect(() => {
   163→    if (typeof window === "undefined" || !window.matchMedia) return;
   164→    const mql = window.matchMedia(query);
   165→    const update = () => setMatches(mql.matches);
   166→    update();
   167→    if (mql.addEventListener) mql.addEventListener("change", update);
   168→    else mql.addListener(update);
   169→    return () => {
   170→      if (mql.removeEventListener) mql.removeEventListener("change", update);
   171→      else mql.removeListener(update);
   172→    };
   173→  }, [query]);
   174→  return matches;
   175→}

   222→function MagneticButton({ children, disabled, reduceMotion, ...rest }) {
   223→  const ref = useRef(null);
   224→  const [offset, setOffset] = useState({ x: 0, y: 0 });
   225→
   226→  const onMove = useCallback(
   227→    (e) => {
   228→      if (reduceMotion || !ref.current) return;
   229→      const r = ref.current.getBoundingClientRect();
   230→      const cx = r.left + r.width / 2;
   231→      const cy = r.top + r.height / 2;
   232→      const dx = e.clientX - cx;
   233→      const dy = e.clientY - cy;
   234→      const max = 4;
   235→      setOffset({
   236→        x: Math.max(-max, Math.min(max, dx * 0.12)),
   237→        y: Math.max(-max, Math.min(max, dy * 0.3)),
   238→      });
   239→    },
   240→    [reduceMotion]
   241→  );
   242→
   243→  const reset = useCallback(() => setOffset({ x: 0, y: 0 }), []);
   244→
   245→  return (
   246→    <button
   247→      ref={ref}
   248→      type="submit"
   249→      disabled={disabled}
   250→      onMouseMove={onMove}
   251→      onMouseLeave={reset}
   252→      className="b13-cta"
   253→      style={{ transform: `translate(${offset.x}px, ${offset.y}px)` }}

   582→
   583→  useEffect(() => {
   584→    // Focus the first cell on mount.
   585→    const first = inputsRef.current[0];
   586→    if (first) first.focus();
   587→  }, []);
   588→
   589→  /* -- DEMO ONLY --------------------------------------------------
   590→     In production, replace this with:
   591→       POST /api/auth/verify-otp   { phone, code }
   592→     and read the resulting session/token + error/rate-limit state
   593→     from the response. OTP generation and checking must never
   594→     happen in the browser. */
   595→  const verify = useCallback(
   596→    (code) => {
   597→      setStatus("verifying");
   598→      setTimeout(() => {
   599→        if (code.length === 6) {
   600→          setStatus("verified");
   601→          // Signature moment: the six cells converge toward the centre
   602→          // and hand off to the success ring before the parent swaps views.
   603→          setTimeout(() => {
   604→            setConverging(true);
   605→            setTimeout(onVerified, reduceMotion ? 0 : 560);
   606→          }, 150);
   607→        } else {
   608→          setStatus("error");
   609→          setErrorShake(true);
   610→          setTimeout(() => setErrorShake(false), 420);
   611→        }
   612→      }, 700);
   613→    },
   614→    [onVerified, reduceMotion]
   615→  );
   616→
   617→  const setDigitAt = (i, val) => {
   618→    setDigits((prev) => {
   619→      const next = [...prev];
   620→      next[i] = val;
   621→      return next;
   622→    });
   623→  };
   624→
   625→  const handleChange = (i, e) => {
   626→    const raw = e.target.value.replace(/[^0-9]/g, "");
   627→    if (!raw) {
   628→      setDigitAt(i, "");
   629→      return;
   630→    }
   631→    const char = raw[raw.length - 1];
   632→    setDigitAt(i, char);
   633→    if (i < 5) {
   634→      const nextInput = inputsRef.current[i + 1];
   635→      if (nextInput) nextInput.focus();
   636→    } else {
   637→      const code = digits.map((d, idx) => (idx === i ? char : d)).join("");
   638→      if (code.length === 6) verify(code);
   639→    }
   640→  };
   641→
   642→  const handleKeyDown = (i, e) => {
   643→    if (e.key === "Backspace") {
   644→      if (digits[i]) {
   645→        setDigitAt(i, "");
   646→      } else if (i > 0) {
   647→        const prevInput = inputsRef.current[i - 1];
   648→        if (prevInput) prevInput.focus();
   649→        setDigitAt(i - 1, "");
   650→      }
   651→    } else if (e.key === "ArrowRight") {
   652→      const t = inputsRef.current[i + 1];
   653→      if (t) t.focus();
   654→    } else if (e.key === "ArrowLeft") {
   655→      const t = inputsRef.current[i - 1];
   656→      if (t) t.focus();
   657→    }
   658→  };
   659→
   660→  const handlePaste = (e) => {
   661→    const text = e.clipboardData.getData("text").replace(/[^0-9]/g, "");
   662→    if (!text) return;
   663→    e.preventDefault();
   664→    const chars = text.slice(0, 6).split("");
   665→    const next = ["", "", "", "", "", ""];
   666→    chars.forEach((c, idx) => (next[idx] = c));
   667→    setDigits(next);
   668→    const lastIdx = Math.min(chars.length, 6) - 1;
   669→    const target = inputsRef.current[lastIdx >= 0 ? lastIdx : 0];
   670→    if (target) target.focus();
   671→    if (chars.length === 6) verify(chars.join(""));
   672→  };
   673→
   674→  const resend = () => {
   675→    setStatus("sent");
   676→    setSeconds(119);
   677→    setDigits(["", "", "", "", "", ""]);
   678→    const first = inputsRef.current[0];
   679→    if (first) first.focus();
   680→  };
   681→
   682→  return (
   683→    <div className="b13-panel-inner">
   684→      <button type="button" className="b13-back" onClick={onBack} aria-label="بازگشت">
   685→        <ArrowRight size={16} strokeWidth={1.5} />
   686→      </button>
   687→
   688→      <div className="b13-panel-head">
   689→        <span className="b13-eyebrow">SECURE ACCESS</span>
   690→        <h1 className="b13-display b13-panel-title">Verify.</h1>
   691→        <p className="b13-fa b13-panel-sub">
   692→          کد تأیید ۶ رقمی برای شماره
   693→          <br />
   694→          <span dir="ltr" className="b13-phone-chip">
   695→            {maskedPhone}
   696→          </span>{" "}
   697→          ارسال شد.
   698→        </p>
   699→      </div>
   700→
   701→      <div
   702→        className={"b13-otp-row" + (errorShake ? " b13-otp-row--shake" : "")}
   703→        onPaste={handlePaste}
   704→        role="group"
   705→        aria-label="کد تأیید شش رقمی"
   706→      >
   707→        {digits.map((d, i) => {
   708→          const dist = i - 2.5;
   709→          const convergeStyle = converging
   710→            ? {
   711→                transform: `translateX(${dist * -30}px) scale(0.35)`,
   712→                opacity: 0,
   713→                transitionDelay: `${Math.abs(dist) * 40}ms`,
   714→                transitionDuration: "480ms",
   715→              }
   716→            : undefined;
   717→          return (
   718→            <input
   719→              key={i}
   720→              ref={(el) => (inputsRef.current[i] = el)}
   721→              type="text"
   722→              inputMode="numeric"
   723→              maxLength={1}
   724→              readOnly={converging}
   725→              value={d}
   726→              onChange={(e) => handleChange(i, e)}
   727→              onKeyDown={(e) => handleKeyDown(i, e)}
   728→              style={convergeStyle}
   729→              className={
   730→                "b13-otp-cell" +
   731→                (d ? " b13-otp-cell--filled" : "") +
   732→                (status === "error" ? " b13-otp-cell--error" : "")
   733→              }
   734→              aria-label={`رقم ${i + 1} از ۶`}
   735→            />
   736→          );
   737→        })}
   738→      </div>
   739→
   740→      <div className="b13-otp-status" aria-live="polite">
   741→        {status === "verifying" && <span className="b13-fa b13-muted">در حال بررسی کد…</span>}
   742→        {status === "verified" && <span className="b13-fa b13-muted">تأیید شد.</span>}
   743→        {status === "error" && <span className="b13-fa b13-error-text">کد وارد شده صحیح نیست.</span>}
   744→        {status === "expired" ? (
   745→          <button type="button" className="b13-fa b13-resend" onClick={resend}>
   746→            ارسال دوباره کد
   747→          </button>
   748→        ) : (
   749→          status !== "verifying" && (
   750→            <span className="b13-fa b13-muted">
   751→              ارسال مجدد کد تا <span dir="ltr">{formatTime(seconds)}</span>
   752→            </span>
   753→          )
   754→        )}
   755→      </div>
   756→    </div>
   757→  );
   758→}

   789→
   790→function AuthCard() {
   791→  const [mode, setMode] = useState("login"); // login | signup
   792→  const [view, setView] = useState("form"); // form | otp | success
   793→  const [rotation, setRotation] = useState(0);
   794→  const [flipping, setFlipping] = useState(false);
   795→  const [morphing, setMorphing] = useState(false);
   796→  const [tilt, setTilt] = useState({ x: 0, y: 0 });
   797→
   798→  const [phone, setPhone] = useState("");
   799→  const [password, setPassword] = useState("");
   800→  const [name, setName] = useState("");
   801→  const [phoneError, setPhoneError] = useState(false);
   802→  const [phoneShake, setPhoneShake] = useState(false);
   803→  const phoneShakeTimer = useRef(null);
   804→
   805→  const reduceMotion = useMediaQuery("(prefers-reduced-motion: reduce)");
   806→  const canHover = useMediaQuery("(hover: hover) and (pointer: fine)");
   807→
   808→  const cardRef = useRef(null);
   809→  const flipTimers = useRef([]);
   810→
   811→  useEffect(() => () => flipTimers.current.forEach(clearTimeout), []);
   812→  useEffect(() => () => clearTimeout(phoneShakeTimer.current), []);
   813→
   814→  const switchMode = useCallback(
   815→    (target) => {
   816→      if (flipping || target === mode) return;
   817→      setPhoneError(false);
   818→      setPhoneShake(false);
   819→      if (reduceMotion) {
   820→        setMode(target);
   821→        return;
   822→      }
   823→      setFlipping(true);
   824→      setRotation((r) => r + 360);
   825→      const t1 = setTimeout(() => setMode(target), ROTATE_MS * SWAP_AT);
   826→      const t2 = setTimeout(() => setFlipping(false), ROTATE_MS);
   827→      flipTimers.current.push(t1, t2);
   828→    },
   829→    [flipping, mode, reduceMotion]
   830→  );
   831→
   832→  const goToOtp = useCallback(() => {
   833→    setMorphing(true);
   834→    setTimeout(() => {
   835→      setView("otp");
   836→      setMorphing(false);
   837→    }, reduceMotion ? 0 : 380);
   838→  }, [reduceMotion]);
   839→
   840→  /* A verification code can only be requested once a phone number has
   841→     been entered — otherwise there is nothing to send the SMS to.
   842→     This guards every path that leads to the OTP screen: the main
   843→     form submit and the "login with code" ghost link alike. */
   844→  const requestOtp = useCallback(() => {
   845→    if (!phone.trim()) {
   846→      setPhoneError(true);
   847→      setPhoneShake(true);
   848→      clearTimeout(phoneShakeTimer.current);
   849→      phoneShakeTimer.current = setTimeout(() => setPhoneShake(false), 420);
   850→      return;
   851→    }
   852→    setPhoneError(false);
   853→    goToOtp();
   854→  }, [phone, goToOtp]);
   855→
   856→  const updatePhone = useCallback((value) => {
   857→    setPhone(value);
   858→    setPhoneError((prev) => (prev && value.trim() ? false : prev));
   859→  }, []);
   860→
   861→  /* DEMO ONLY: In production, replace this with a real "Sign in with
   862→     Google" flow (e.g. Google Identity Services / OAuth redirect),
   863→     then call goToSuccess() once the provider confirms the session.
   864→     Google auth doesn't need the phone/OTP step, so it goes straight
   865→     to success. */
   866→  const continueWithGoogle = useCallback(() => {
   867→    setMorphing(true);
   868→    setTimeout(() => {
   869→      setView("success");
   870→      setMorphing(false);
   871→    }, reduceMotion ? 0 : 380);
   872→  }, [reduceMotion]);
   873→
   874→  const backToForm = useCallback(() => {
   875→    setMorphing(true);
   876→    setTimeout(() => {
   877→      setView("form");
   878→      setMorphing(false);
   879→    }, reduceMotion ? 0 : 380);
   880→  }, [reduceMotion]);
   881→
   882→  const goToSuccess = useCallback(() => {
   883→    setView("success");
   884→  }, []);
   885→
   886→  const resetAll = useCallback(() => {
   887→    setView("form");
   888→    setMode("login");
   889→    setPhone("");
   890→    setPassword("");
   891→    setName("");
   892→    setPhoneError(false);
   893→    setPhoneShake(false);
   894→  }, []);
   895→
   896→  const onLoginSubmit = (e) => {
   897→    e.preventDefault();
   898→    requestOtp();
   899→  };
   900→  const onSignupSubmit = (e) => {
   901→    e.preventDefault();
   902→    requestOtp();
   903→  };
   904→
   905→  const onCardMouseMove = (e) => {
   906→    if (!canHover || reduceMotion || flipping || !cardRef.current) return;
   907→    const r = cardRef.current.getBoundingClientRect();
   908→    const px = (e.clientX - r.left) / r.width - 0.5;
   909→    const py = (e.clientY - r.top) / r.height - 0.5;
   910→    setTilt({ x: py * -2, y: px * 2 });
   911→  };
   912→  const onCardMouseLeave = () => setTilt({ x: 0, y: 0 });
   913→
   914→  const stage = view === "otp" ? "otp" : view === "success" ? "success" : mode;
   915→  const markLines = view === "otp" ? ["0", "13"] : view === "success" ? ["13"] : COPY[mode].mark;
   916→
   917→  return (
   918→    <div className="b13-scene">
   919→      <AmbientBackground sweeping={flipping} />
   920→      <Mark13 lines={markLines} active={flipping} />
   921→
   922→      <div
   923→        className="b13-card-perspective"
   924→        onMouseMove={onCardMouseMove}
   925→        onMouseLeave={onCardMouseLeave}
   926→      >
   927→        <div
   928→          ref={cardRef}
   929→          className={"b13-card" + (flipping ? " b13-card--flipping" : "")}
   930→          style={{
   931→            transform: `rotateY(${rotation}deg) rotateX(${tilt.x}deg) rotateY(${tilt.y}deg)`,
   932→          }}
   933→        >
   934→          <div className={"b13-card-surface" + (morphing ? " b13-card-surface--morph" : "")}>
   935→            <div className="b13-grid">
   936→              <VisualPanel stage={stage} tone={mode === "login" ? "a" : "b"} />
   937→
   938→              <div className="b13-form-panel">
   939→                {view === "form" && mode === "login" && (
   940→                  <LoginView
   941→                    phone={phone}
   942→                    setPhone={updatePhone}
   943→                    password={password}
   944→                    setPassword={setPassword}
   945→                    onSubmit={onLoginSubmit}
   946→                    onOtpRequest={requestOtp}
   947→                    onSwitch={() => switchMode("signup")}
   948→                    onGoogleAuth={continueWithGoogle}
   949→                    phoneError={phoneError}
   950→                    phoneShake={phoneShake}
   951→                    reduceMotion={reduceMotion}
   952→                  />
   953→                )}
   954→                {view === "form" && mode === "signup" && (
   955→                  <SignupView
   956→                    name={name}
   957→                    setName={setName}
   958→                    phone={phone}
   959→                    setPhone={updatePhone}
   960→                    password={password}
   961→                    setPassword={setPassword}
   962→                    onSubmit={onSignupSubmit}
   963→                    onSwitch={() => switchMode("login")}
   964→                    onGoogleAuth={continueWithGoogle}
   965→                    phoneError={phoneError}
   966→                    phoneShake={phoneShake}
   967→                    reduceMotion={reduceMotion}
   968→                  />
   969→                )}
   970→                {view === "otp" && (
   971→                  <OtpView phone={phone} onBack={backToForm} onVerified={goToSuccess} reduceMotion={reduceMotion} />
   972→                )}
   973→                {view === "success" && <SuccessView onEnter={resetAll} reduceMotion={reduceMotion} />}
   974→              </div>
   975→            </div>
   976→          </div>
   977→        </div>
   978→      </div>
   979→    </div>
   980→  );
   981→}
